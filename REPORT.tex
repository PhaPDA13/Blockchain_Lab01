\chapter{Giới thiệu và Cấu trúc dự án}

\section{Mục tiêu và Bối cảnh}
Hệ thống blockchain phân tán dựa trên mạng ngang hàng (P2P) hoạt động trong môi trường không tin cậy đòi hỏi sự phối hợp chặt chẽ giữa tính toàn vẹn dữ liệu mật mã, sự lan truyền thông điệp mạng, cơ chế đồng thuận kháng lỗi Byzantine (BFT) và thực thi trạng thái xác định. Dự án này hiện thực một mô hình blockchain Layer 1 thu nhỏ hoàn chỉnh, hướng đến việc đạt được tính chốt khối tin cậy (\emph{reliable finality}) dưới điều kiện mạng truyền thông điệp không tin cậy (chịu ảnh hưởng bởi trễ, mất gói, trùng lặp và xáo trộn thứ tự).

Mục tiêu cốt lõi bao gồm:
\begin{itemize}
  \item \textbf{Tính xác thực và toàn vẹn:} Mỗi giao dịch, đề xuất khối và phiếu bầu đều được bảo vệ bằng mật mã chữ ký số Ed25519, hàm băm SHA-256 và domain separation nghiêm ngặt.
  \item \textbf{Thực thi xác định và chống phát lại:} State chuyển đổi xác định theo mô hình key-value; kiểm tra quyền sở hữu tiền tố người gửi và nonce để từ chối giao dịch đã thực thi hoặc sai thứ tự.
  \item \textbf{Đồng thuận Byzantine Fault Tolerant (BFT):} Vận hành giao thức đồng thuận hai pha (Prevote và Precommit) tuân thủ 7 quy tắc khóa khối (Locking Rules), hướng tới bảo toàn tính an toàn (\emph{Safety}) theo mô hình lỗi đã nêu và tính sống (\emph{Liveness}) sau khi mạng ổn định, khi proposer trung thực được chọn.
  \item \textbf{Mô phỏng mạng rời rạc xác định:} Mô hình hóa mạng bán đồng bộ (\emph{partially synchronous network}) có khả năng tái lập vết sự kiện (\emph{byte-identical logs}) thông qua hạt giống ngẫu nhiên (\emph{seed}).
\end{itemize}

\section{Các quyết định thiết kế cốt lõi}
Hệ thống được thiết kế dựa trên các nguyên tắc kỹ thuật sau:
\begin{enumerate}
  \item \textbf{Domain Separation tường minh:} Ngăn ngừa tấn công chuyển tiếp thông điệp chéo (\emph{cross-protocol replay}) bằng các tiền tố ngữ cảnh: \texttt{TX:chain\_id}, \texttt{HEADER:chain\_id}, \texttt{VOTE:chain\_id}.
  \item \textbf{Mã hóa Canonical JSON:} Mọi cấu trúc dữ liệu trước khi băm hoặc ký đều được tuần tự hóa xác định (sắp xếp khóa từ điển, bỏ khoảng trắng dư thừa, mã hóa UTF-8).
  \item \textbf{Tách biệt Header và Body:} Giảm thiểu băng thông lãng phí bằng cách broadcast \texttt{HEADER} trước; các node chỉ gửi \texttt{BODY\_REQUEST} và nhận \texttt{BODY} khi header hợp lệ.
  \item \textbf{Tách biệt Semantic Block ID và Proposal Hash:} Mã băm định danh nội dung khối (\texttt{block\_id}) độc lập với thông tin vòng/người đề xuất (\texttt{header\_hash}), cho phép proposer đề xuất lại khối đã khóa ở các round tiếp theo mà không làm sai lệch trạng thái khóa.
  \item \textbf{Chứng thực chốt khối di động (Finality Certificate):} Cho phép các node bị trễ hoặc khởi động lại bắt kịp chiều cao chuỗi khối tức thời mà không cần trải qua toàn bộ quá trình bỏ phiếu cục bộ.
\end{enumerate}

\section{Cấu trúc thư mục dự án}
Dự án được cấu trúc theo đúng quy định tại Mục 10 của đề bài:
\begin{description}[leftmargin=1.2em, itemsep=1.5pt]
  \item[\texttt{src/crypto/}] Module mật mã: mã hóa chuẩn tắc \texttt{encoding.py}, hàm băm SHA-256 \& state root \texttt{hashing.py}, quản lý khóa Ed25519 \& chữ ký domain \texttt{identity.py}.
  \item[\texttt{src/execution/}] Module thực thi: cấu trúc giao dịch \texttt{transaction.py}, khối \& tiêu đề \texttt{block.py}, sổ cái trạng thái key-value \& thực thi nguyên tử \texttt{state.py}.
  \item[\texttt{src/consensus/}] Module đồng thuận: định nghĩa phiếu bầu \texttt{vote.py}, 7 quy tắc khóa khối \texttt{rules.py}, động cơ đồng thuận hai pha \& quorum \texttt{engine.py}.
  \item[\texttt{src/network/}] Module mạng mô phỏng: định dạng thông điệp \texttt{message.py}, hàng đợi trễ/mất gói \texttt{simulator.py}, bộ ghi vết \texttt{logger.py}, vòng lặp sự kiện \texttt{runner.py}.
  \item[\texttt{src/node.py}] Node validator tích hợp toàn diện các tầng thực thi, đồng thuận và bộ đệm giao dịch.
  \item[\texttt{tests/}] Bộ kiểm thử gồm \texttt{tests/unit/} và \texttt{tests/e2e/test\_scenarios.py} (kịch bản T1--T8 và hồi quy).
  \item[\texttt{config/}] Chứa tệp cấu hình mạng mặc định \texttt{default\_config.json}.
  \item[\texttt{logs/}] Thư mục lưu vết log mô phỏng xác định (canonical byte logs).
  \item[\texttt{run\_simulation.py}] Script thực thi mô phỏng 1 khối hoàn chỉnh trên mạng validator.
  \item[\texttt{package\_submission.py}] Công cụ tự động đóng gói bài nộp theo định dạng chuẩn quy định.
  \item[\texttt{README.md}, \texttt{REPORT.pdf}] Tài liệu hướng dẫn cài đặt/chạy kiểm thử và báo cáo kỹ thuật.
\end{description}

\section{Mô hình lỗi và Giả định mạng}
\begin{itemize}
  \item \textbf{Mô hình validator:} Tập hợp validator cố định $n$ node với $n \ge 8$ và thỏa $n = 3f + 1$, trong đó có tối đa $f = \lfloor(n - 1) / 3\rfloor$ node Byzantine. Với cấu hình mặc định $n = 10$, hệ thống chịu đựng được $f = 3$ node lỗi; ngưỡng quorum đạt $2f + 1 = 7$ phiếu.
  \item \textbf{Hành vi Byzantine:} Validator Byzantine có thể im lặng (silent/crash), gửi tin nhắn lặp (duplicate), bỏ phiếu xung đột cho nhiều khối khác nhau (equivocation), đề xuất khối không hợp lệ hoặc phát lại tin cũ. Validator Byzantine không thể giả mạo chữ ký của node trung thực hay tìm va chạm hàm băm SHA-256.
  \item \textbf{Mô hình mạng bán đồng bộ:} Trước thời điểm ổn định ($T_{stabilization}$), mạng có thể trễ tùy ý, làm rơi, nhân bản hoặc đảo thứ tự thông điệp. Sau $T_{stabilization}$, mọi thông điệp giữa các node trung thực được chuyển giao trong cận trễ $\Delta$ (\texttt{delta\_bound\_ticks}).
\end{itemize}

Chi tiết các bước thực hiện:
\begin{enumerate}
  \item \textbf{Ingress Giao dịch:} Node ingress nhận giao dịch, kiểm tra chữ ký và tính hợp lệ trước khi đưa vào mempool cục bộ và gossip sang các peer khác.
  \item \textbf{Đề xuất Khối (Proposal):} Proposer xác định theo lịch vòng tròn ($idx = (height + round) \pmod n$). Proposer trích xuất các giao dịch hợp lệ, thực thi trên bản sao trạng thái (\emph{state copy}) để tính \texttt{state\_root}, đóng gói \texttt{transactions\_root} và ký phát tán \texttt{HEADER}.
  \item \textbf{Đóng khung Header -- Body:} Node nhận xác thực chữ ký proposer, parent hash và lịch round. Khi header hợp lệ, node gửi tin \texttt{BODY\_REQUEST} tới proposer để nhận \texttt{BODY}.
  \item \textbf{Pha PREVOTE:} Sau khi nhận đủ body, node thực thi tuần tự các transaction trên state copy. Nếu kết quả trùng khớp với \texttt{state\_root} trong header, node ký phát tán \texttt{PREVOTE} cho mã băm khối; ngược lại bỏ phiếu cho NIL.
  \item \textbf{Khóa khối và Pha PRECOMMIT:} Khi quan sát thấy $\ge 2f + 1$ phiếu PREVOTE cho khối hợp lệ, node thực hiện khóa (\emph{lock}) khối đó tại round hiện tại và ký phát tán \texttt{PRECOMMIT}.
  \item \textbf{Chốt khối (Finalization):} Khi quan sát thấy $\ge 2f + 1$ phiếu PRECOMMIT cho cùng một khối, node áp dụng vĩnh viễn các giao dịch vào trạng thái chính thức, tăng chiều cao khối (\texttt{height += 1}), thiết lập lại khóa và phát tán chứng thực chốt khối (\texttt{FINALIZED}).
\end{enumerate}
\medskip\noindent\textbf{Cơ chế điều phối vòng lặp sự kiện (\texttt{NetworkRunner} \& \texttt{NetworkSimulator}):}
Toàn bộ quá trình tương tác giữa các validator được vận hành theo thời gian logic rời rạc (\emph{discrete ticks}). Tại mỗi bước thời gian (\texttt{tick}), \texttt{NetworkRunner} điều phối theo thứ tự xác định: (1) kích hoạt proposer của vòng hiện tại tạo đề xuất khối nếu chưa phát tán, (2) chuyển giao các thông điệp đã đến hạn trong hàng đợi của \texttt{NetworkSimulator} tới đúng hàm xử lý của từng \texttt{Node}, (3) kiểm tra ngưỡng quorum để chuyển pha từ \texttt{PREVOTE} sang \texttt{PRECOMMIT} hoặc chốt khối (\texttt{finalize\_block}), và (4) kiểm tra thời hạn \texttt{round\_timeout\_ticks} cũng như chu kỳ phát lại \texttt{retransmit\_interval\_ticks}.

\chapter{Hiện thực và Cơ chế bảo vệ}

\section{Cơ chế bảo vệ Cryptography và Domain Separation}
Transaction, block header và vote được ký bằng Ed25519 trong các domain riêng để ngăn việc tái sử dụng chữ ký giữa loại thông điệp hoặc chain khác nhau. Thông điệp \texttt{FINALIZED} mang header đã ký cùng certificate gồm các vote PRECOMMIT đã ký; các envelope truyền tải khác không được ký độc lập. Hệ thống áp dụng các domain sau:
\begin{itemize}
  \item \textbf{Giao dịch:} Ký trên ngữ cảnh \texttt{TX:<chain\_id>} với payload gồm \texttt{\{sender, key, value, nonce\}}.
  \item \textbf{Tiêu đề khối:} Ký trên ngữ cảnh \texttt{HEADER:<chain\_id>} với payload gồm \texttt{\{parent\_hash, height, round, state\_root, proposer, transactions\_root\}}.
  \item \textbf{Phiếu bầu:} Ký trên ngữ cảnh \texttt{VOTE:<chain\_id>} với payload gồm \texttt{\{validator, height, round, phase, block\_hash\}}.
\end{itemize}
Mọi payload được đưa về chuỗi byte chuẩn hóa bằng \texttt{canonical\_encode} trước khi đưa vào hàm ký hoặc băm, đảm bảo tính nhất quán trên các nền tảng khác nhau.

\section{Bảo vệ Trạng thái và Thực thi giao dịch}
Trạng thái ứng dụng được quản lý dưới dạng bảng ánh xạ key-value trong \texttt{State}. Cơ chế bảo vệ thực thi bao gồm:
\begin{itemize}
  \item \textbf{Kiểm tra quyền sở hữu khóa (Ownership Guard):} Khóa trạng thái bắt buộc phải có tiền tố là địa chỉ công khai của người gửi (\texttt{key.startswith(sender + "/")}). Không một tài khoản nào có thể thay đổi dữ liệu của tài khoản khác.
  \item \textbf{Chống phát lại bằng Nonce tuần tự:} Mỗi tài khoản duy trì một bộ đếm nonce bắt đầu từ 0. Giao dịch chỉ được chấp nhận nếu $nonce == expected\_nonce$. Sau khi thực thi thành công, nonce của tài khoản tăng lên 1 và hash của giao dịch được lưu vào tập \texttt{\_executed\_tx\_hashes}.
  \item \textbf{Thực thi nguyên tử (All-or-Nothing):} Khi thực thi một khối, toàn bộ danh sách giao dịch được áp dụng trên bản sao chụp trạng thái (\emph{snapshot}). Nếu có bất kỳ giao dịch nào không hợp lệ, toàn bộ khối bị từ chối và trạng thái gốc được bảo toàn.
  \item \textbf{Cam kết State Root toàn diện:} \texttt{state\_root} là hàm băm SHA-256 cam kết đồng thời cả dữ liệu bản ghi (\texttt{records}) lẫn trạng thái nonce của người gửi (\texttt{sender\_nonces}), ngăn chặn việc sai lệch lịch sử giao dịch.
\end{itemize}

\section{Bảy quy tắc đồng thuận và Khóa khối (Mục 6.1)}
Lớp đồng thuận cài đặt nghiêm ngặt 7 quy tắc khóa khối trong \texttt{ConsensusLockState}:
\begin{enumerate}
  \item Ký tối đa một phiếu \texttt{PREVOTE} cho mỗi cặp \texttt{(height, round)}.
  \item Ký tối đa một phiếu \texttt{PRECOMMIT} cho mỗi cặp \texttt{(height, round)}.
  \item Chỉ chấp nhận và bỏ phiếu PREVOTE cho khối có parent hợp lệ, giao dịch hợp lệ, state root chính xác, đúng proposer và chữ ký proposer hợp lệ.
  \item Khi quan sát thấy quorum PREVOTE ($\ge 2f + 1$) cho một khối non-NIL, validator khóa khối đó và ghi nhận round khóa tương ứng (\texttt{locked\_round}).
  \item Validator đang khóa khối $B$ tại vòng $r_{locked}$ (\texttt{locked\_round}) chỉ được phép mở khóa để PREVOTE cho khối $B' \neq B$ tại vòng hiện tại $r_{current}$ nếu đã quan sát thấy một quorum PREVOTE hợp lệ cho $B'$ ở vòng $r_{quorum}$ thỏa mãn bất đẳng thức $r_{locked} < r_{quorum} < r_{current}$ (hiện thực qua \texttt{can\_prevote\_block} và \texttt{\_latest\_prevote\_quorum\_round}). Nếu bản thân validator đến lượt làm proposer khi đang khóa khối $B$, node ưu tiên đề xuất lại nội dung khối $B$ với tiêu đề gắn số vòng mới.
  \item Validator từ chối ký phiếu bầu sai chain, height, round, phase hoặc sai domain.
  \item Bỏ qua phiếu bầu trùng lặp; chỉ đếm tối đa một phiếu hợp lệ từ mỗi validator cho mỗi bộ ba \texttt{(height, round, phase)}.
\end{enumerate}

\section{Chiến lược mạng và Đồng bộ phục hồi}
\begin{itemize}
  \item \textbf{Giới hạn thông lượng (Rate Limiting):} Simulator theo dõi thời điểm gửi thông điệp của từng node trong cửa sổ \texttt{rate\_window\_ticks}. Nếu vượt quá \texttt{max\_outbound\_rate}, thông điệp bị hủy và ghi log sự kiện \texttt{drop}.
  \item \textbf{Cô lập ngang hàng (Peer Blocking):} Hỗ trợ cô lập hai chiều giữa hai node thông qua \texttt{block\_peer} và phục hồi lại bằng \texttt{unblock\_peer}.
  \item \textbf{Quản lý Timeout tăng dần:} Thời gian chờ của round tăng tuyến tính theo số vòng thất bại ($timeout = base \times (round + 1)$) nhằm dành thêm thời gian cho thông điệp ở các round sau.
  \item \textbf{Cơ chế bắt kịp qua Certificate:} Khi một node nhận được thông điệp \texttt{FINALIZED} kèm danh sách $\ge 2f + 1$ phiếu PRECOMMIT hợp lệ, node có thể chốt khối và nhảy cóc lên height mới mà không bị kẹt ở round cũ.
  \item \textbf{Chiến lược phát tán (Gossip) và Phát lại định kỳ (Retransmission):} Giao dịch hợp lệ tại node ingress được phát tán ngang hàng sang toàn mạng và lọc trùng bằng mã băm \texttt{tx\_hash}. Để khắc phục hiện tượng mất gói trước thời điểm ổn định $T_{stabilization}$, \texttt{NetworkRunner} duy trì cơ chế phát lại định kỳ mỗi \texttt{retransmit\_interval\_ticks} đối với các đề xuất khối và phiếu bầu còn hiệu lực, kết hợp với đóng khung hai bước (\texttt{HEADER} $\rightarrow$ \texttt{BODY\_REQUEST} $\rightarrow$ \texttt{BODY}) để tiết kiệm hạn ngạch băng thông.
  \item \textbf{Ghi vết sự kiện xác định (Deterministic Canonical Logging):} Mọi sự kiện mạng (\texttt{send}, \texttt{drop}, \texttt{delay}, \texttt{duplicate}, \texttt{block}, \texttt{unblock}) và sự kiện đồng thuận của node đều được \texttt{NetworkLogger} ghi nhận kèm \texttt{timestamp} (tick logic), \texttt{node\_id}, \texttt{type}, \texttt{height} và \texttt{details}. Toàn bộ danh sách sự kiện được tuần tự hóa bằng \texttt{export\_canonical\_bytes()} giúp kiểm chứng đối sánh trùng khớp từng byte giữa các lần chạy.
\end{itemize}

\section{Các lỗi phi tầm thường phát hiện và Cách giải quyết}
Trong quá trình phát triển và kiểm thử sâu rộng hệ thống, nhóm đã phát hiện và xử lý 5 lỗi kỹ thuật phi tầm thường (\emph{non-trivial bugs}):
\begin{enumerate}
  \item \textbf{Lỗi reset timer vòng đồng thuận ở chiều cao mới (Critical):}
  \emph{Hiện tượng:} Khi validator hoàn tất chốt khối ở height 1 tại tick 18 và chuyển sang height 2, bộ đếm thời gian round của height 2 được khởi tạo mặc định bằng 0 (\texttt{\_round\_started.setdefault(key, 0)}). Do đó, phép so sánh $now - started = 18 - 0 = 18 \ge timeout$ lập tức kích hoạt timeout liên tục, khiến node nhảy round bất tận và không thể chốt các khối tiếp theo.
  \emph{Giải pháp:} Khởi tạo thời điểm bắt đầu round bằng thời gian logic hiện tại của simulator: \texttt{\_round\_started.setdefault(key, self.simulator.current\_time)}.

  \item \textbf{Từ chối Header/Body đến trễ sau khi đã chuyển round (High):}
  \emph{Hiện tượng:} Khi một node bị trễ cục bộ và đã timeout sang round $r + 1$, các gói tin \texttt{HEADER} và \texttt{BODY} hợp lệ được đề xuất ở round $r$ bị hàm kiểm tra từ chối do mặc định tham số \texttt{require\_current\_round = True}. Điều này làm node không thể thu thập đủ bằng chứng khối để tham gia vào quorum precommit.
  \emph{Giải pháp:} Cho phép tiếp nhận và lưu trữ khối đề xuất hợp lệ từ các round trước đó bằng cách đặt \texttt{require\_current\_round = False} đối với các khối có $round \le current\_round$.

  \item \textbf{State Root không cam kết trạng thái Nonce (Medium):}
  \emph{Hiện tượng:} Ban đầu \texttt{compute\_state\_root} chỉ băm danh sách các bản ghi key-value. Nếu hai trạng thái có cùng dữ liệu nhưng khác số nonce của tài khoản, mã băm state root vẫn giống nhau, tạo lỗ hổng cho phép tái sử dụng giao dịch ở các nhánh rẽ.
  \emph{Giải pháp:} Cập nhật hàm tính state root để băm đồng thời cả cấu trúc \texttt{records} và \texttt{sender\_nonces} dưới dạng chuẩn tắc.

  \item \textbf{Đồng nhất hóa định danh khối giữa các vòng (Medium):}
  \emph{Hiện tượng:} Ban đầu mã băm khối \texttt{block\_hash} được tính trực tiếp từ \texttt{header\_hash}. Khi proposer ở round mới đề xuất lại khối đã khóa từ round trước, tiêu đề khối bắt buộc phải chứa round mới và chữ ký mới, dẫn đến \texttt{block\_hash} bị thay đổi và vi phạm quy tắc khóa khối (Rule 5).
  \emph{Giải pháp:} Tách biệt \texttt{header\_hash} (phục vụ xác thực đề xuất cụ thể của round) và \texttt{block\_id} (cam kết ngữ nghĩa gồm parent hash, height, state root và transactions root) để bảo toàn danh tính khối qua các round.

  \item \textbf{Rò rỉ đối tượng Vote có thể sửa đổi (Low):}\\
  \emph{Hiện tượng:} Đối tượng \texttt{Vote} là mutable dataclass. Khi \texttt{ConsensusLockState} lưu trữ và trả về tham chiếu trực tiếp, các module bên ngoài có thể vô tình hoặc cố ý thay đổi trường dữ liệu phiếu bầu, làm sai lệch kết quả kiểm tra quorum nội bộ.
  \emph{Giải pháp:} Cài đặt sao chép phòng vệ (\emph{defensive copy}) khi tiếp nhận và truy xuất phiếu bầu trong \texttt{ConsensusLockState}.
\end{enumerate}

\section{Phân tích các tình huống dị thường}

Trong kịch bản cô lập này, ba phiếu Byzantine không tạo thành quorum $2f + 1 = 7$; kết quả xác nhận lọc vote trùng, không thay thế kiểm thử đồng thuận đầy đủ có Byzantine.
  \item \textbf{Phân tích mất gói và phục hồi tính sống (T5):} Trước tick 16, xác suất mất gói được thiết lập lên tới 40\% và trễ tối đa 4 tick. Các vòng đề xuất ban đầu timeout do thiếu thông điệp; không có hai khối xung đột cùng được chốt. Sau khi mạng ổn định, cả 10 node chốt cùng khối tại round 2, tick 34 với cận trễ $\Delta = 2$ tick.

Kiểm tra so sánh đối chiếu mảng byte (\texttt{bytes equality}) giữa hai lần chạy cho thấy: chuỗi khối đã chốt, mã băm state root, toàn bộ log sự kiện của từng validator và log sự kiện của simulator giống nhau tuyệt đối từng byte.
\end{itemize}

\section{Kết quả đạt được}

\begin{itemize}
  \item Xây dựng hoàn chỉnh một validator node Layer 1 với kiến trúc phân tầng khoa học, tách bạch giữa mật mã, thực thi trạng thái, cơ chế đồng thuận và mô phỏng mạng.
  \item Hiện thực giao thức đồng thuận BFT hai pha theo 7 quy tắc khóa khối và kiểm tra các kịch bản Byzantine/mạng lỗi đã chọn; các bài kiểm thử hữu hạn không thay thế một chứng minh hình thức cho mọi lịch mạng.
  \item Hiện thực cơ chế thực thi trạng thái an toàn với kiểm tra quyền sở hữu, số thứ tự nonce chống phát lại và cam kết state root xác định.
  \item Xây dựng bộ kiểm thử tự động gồm 35 test cho các thành phần chính và các kịch bản dị thường đã chọn, kèm khả năng tái lập vết sự kiện bằng log canonical.
\end{itemize}

\section{Hạn chế và Hướng mở rộng}
\begin{itemize}
  \item \textbf{Cấu trúc dữ liệu State:} Hiện tại state root sử dụng hàm băm SHA-256 trên toàn bộ từ điển đã sắp xếp. Để tối ưu hóa hiệu năng khi số lượng tài khoản lớn, hệ thống có thể mở rộng nâng cấp sang cấu trúc Merkle Patricia Trie (MPT) hoặc Verkle Tree nhằm cung cấp bằng chứng thành viên (\emph{Merkle proof}) cho các light client.
  \item \textbf{Giới hạn của lớp mạng mô phỏng:} \texttt{NetworkRunner} điều phối tập trung, dùng cache block chung và dựa vào height của toàn mạng khi phát lại certificate; do đó chưa mô phỏng hoàn toàn trạng thái cô lập của từng node. Chặn peer hiện phải kích hoạt thủ công, vote từ round tương lai bị từ chối nên node trễ phải chờ timeout. Rate limit và retransmission mặc định cũng có thể làm chậm các height tiếp theo; chưa có kiểm thử multi-height với cấu hình mặc định.
  \item \textbf{Cơ chế đồng bộ trạng thái nhanh (Fast Sync):} Nghiên cứu tích hợp snapshot trạng thái và đồng bộ hóa qua mạng đối với các node mới tham gia hoặc các node bị gián đoạn hoạt động trong thời gian dài.
  \item \textbf{Mô hình giao dịch mở rộng:} Bổ sung hệ thống tính toán phí giao dịch (gas fee) và hỗ trợ thực thi hợp đồng thông minh tối giản trên máy ảo phân tán.
\end{itemize}
